== COMPILAR (PowerShell) ==
gcc -O2 -Wall -Wextra -Wpedantic -std=c11 -o bin/matmul src/matmul.c -lm
gcc -O2 -Wall -Wextra -std=c11 -pthread -o bin/mmHILOS src/mmHILOS.c -lm


== 1) SECUENCIAL: 10 turnos x (500, 1000, 1500) ==
"Orden;Tiempo(s);Rendimiento(GOP/s)" | Out-File resultados_secuencial.csv -Encoding utf8; 1..10 | ForEach-Object { foreach ($n in 500,1000,1500) { .\bin\matmul.exe -n $n -l 100 -c | Out-File resultados_secuencial.csv -Encoding utf8 -Append } }


== 2) HILOS: 10 turnos x (500, 1000, 1500) x (2, 4, 8, 16 hilos) ==
"Orden;NumHilos;Tiempo(s);Rendimiento(GOP/s)" | Out-File resultados_hilos.csv -Encoding utf8; 1..10 | ForEach-Object { foreach ($n in 500,1000,1500) { foreach ($t in 2,4,8,16) { .\bin\mmHILOS.exe -n $n -l 100 -t $t -c | Out-File resultados_hilos.csv -Encoding utf8 -Append } } }
